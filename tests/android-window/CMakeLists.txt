cmake_minimum_required(VERSION 3.20)
project(android_window C)
if(MALVADO_TEST_SDK)
  find_package(Malvado CONFIG REQUIRED COMPONENTS graphics
    PATHS "${MALVADO_TEST_SDK}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
else()
  set(MALVADO_BUILD_LUA OFF CACHE BOOL "" FORCE)
  set(MALVADO_BUILD_ASSETS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
endif()
add_library(game SHARED main.c)
if(MALVADO_TEST_SDK)
  target_link_libraries(game PRIVATE Malvado::graphics)
else()
  target_link_libraries(game PRIVATE malvado_platform_raylib)
endif()
target_link_options(game PRIVATE "-Wl,--no-undefined")
if(NOT MALVADO_TEST_SDK)
  target_link_options(game PRIVATE "-uANativeActivity_onCreate")
endif()
target_compile_definitions(game PRIVATE ACCEPTANCE_TOKEN="${ACCEPTANCE_TOKEN}")

target_compile_definitions(game PRIVATE ACCEPTANCE_LIFECYCLE=${ACCEPTANCE_LIFECYCLE})
target_include_directories(game PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")

target_compile_definitions(game PRIVATE ACCEPTANCE_AUDIO=${ACCEPTANCE_AUDIO})
