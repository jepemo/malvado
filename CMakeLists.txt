cmake_minimum_required(VERSION 3.20)
project(malvado VERSION 2.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
include(FetchContent)
include(GNUInstallDirs)
option(MALVADO_BUILD_LUA "Build the Lua runtime" ON)
option(MALVADO_BUILD_RAYLIB "Build the raylib backend" ON)

add_library(malvado_core STATIC src/core/clock.c src/core/schedule.c src/core/context.c src/core/process.c)
target_include_directories(malvado_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src/core)
if(UNIX)
  target_link_libraries(malvado_core PUBLIC m)
endif()
install(TARGETS malvado_core ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/malvado DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

add_executable(example-c-clock examples/c/clock/main.c)
target_link_libraries(example-c-clock PRIVATE malvado_core)
add_executable(example-c-processes examples/c/processes/main.c)
target_link_libraries(example-c-processes PRIVATE malvado_core)

if(MALVADO_BUILD_RAYLIB)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(raylib)
  add_library(malvado_platform_raylib STATIC src/platform/raylib.c)
  target_include_directories(malvado_platform_raylib PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(malvado_platform_raylib PUBLIC raylib)
  install(TARGETS malvado_platform_raylib ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  add_executable(example-c-window examples/c/window/main.c)
  target_link_libraries(example-c-window PRIVATE malvado_core malvado_platform_raylib)
else()
  add_library(malvado_platform_headless STATIC src/platform/headless.c)
  target_include_directories(malvado_platform_headless PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
endif()
if(MALVADO_BUILD_LUA)
  FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(lua)
  file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${lua_SOURCE_DIR}/src/*.c")
  list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
  add_library(malvado_lua STATIC ${LUA_SOURCES})
  target_include_directories(malvado_lua PUBLIC "${lua_SOURCE_DIR}/src")
  if(UNIX)
    target_compile_definitions(malvado_lua PRIVATE LUA_USE_POSIX)
    target_link_libraries(malvado_lua PUBLIC m)
  endif()
  add_library(malvado_lua_binding STATIC src/binding/runtime.c)
  target_include_directories(malvado_lua_binding PRIVATE src)
  if(MALVADO_BUILD_RAYLIB)
    target_link_libraries(malvado_lua_binding PUBLIC malvado_lua malvado_core malvado_platform_raylib)
  else()
    target_link_libraries(malvado_lua_binding PUBLIC malvado_lua malvado_core malvado_platform_headless)
  endif()
  add_executable(malvado-run src/main.c)
  target_include_directories(malvado-run PRIVATE src)
  target_link_libraries(malvado-run PRIVATE malvado_lua_binding)
endif()
if(MSVC)
  if(TARGET malvado-run)
    target_compile_options(malvado-run PRIVATE /W4)
    target_compile_options(malvado_lua_binding PRIVATE /W4)
  endif()
  target_compile_options(malvado_core PRIVATE /W4)
else()
  if(TARGET malvado-run)
    target_compile_options(malvado-run PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(malvado_lua_binding PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  target_compile_options(malvado_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
