cmake_minimum_required(VERSION 3.20)
project(malvado_rust_probe C)
set(CMAKE_C_STANDARD 11)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
unset(Malvado_DIR CACHE)
set(components core)
if(MALVADO_RUST_ASSETS)
  list(APPEND components assets)
endif()
if(MALVADO_RUST_GRAPHICS)
  list(APPEND components graphics)
endif()
find_package(Malvado CONFIG REQUIRED COMPONENTS ${components}
  PATHS "${MALVADO_SDK_PREFIX}" NO_DEFAULT_PATH)
get_target_property(core_type Malvado::core TYPE)
if(NOT core_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Rust currently requires a static Malvado core SDK")
endif()
add_executable(probe probe.c)
target_link_libraries(probe PRIVATE Malvado::core)
target_compile_definitions(probe PRIVATE EXPECTED_VERSION="${EXPECTED_VERSION}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/native-library.txt" CONTENT "$<TARGET_FILE:Malvado::core>")

if(MALVADO_RUST_ASSETS)
  get_target_property(assets_type Malvado::assets TYPE)
  if(NOT assets_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Rust assets feature requires static Malvado assets")
  endif()
  add_executable(assets-probe assets-probe.c)
  target_link_libraries(assets-probe PRIVATE Malvado::assets)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/assets-library.txt" CONTENT "$<TARGET_FILE:Malvado::assets>")
endif()

if(MALVADO_RUST_GRAPHICS)
  get_target_property(graphics_type Malvado::graphics TYPE)
  if(NOT graphics_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Rust graphics feature requires static Malvado graphics")
  endif()
  # Native thread checks and verification helpers share the graphics archive.
  add_library(graphics-checks STATIC graphics-checks.c)
  if(WIN32)
    target_sources(graphics-checks PRIVATE windows-thread.c)
  endif()
  target_link_libraries(graphics-checks PRIVATE Malvado::graphics)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/graphics-checks-library.txt" CONTENT "$<TARGET_FILE:graphics-checks>")
  add_executable(graphics-probe graphics-probe.c)
  target_link_libraries(graphics-probe PRIVATE Malvado::graphics)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/graphics-library.txt" CONTENT "$<TARGET_FILE:Malvado::graphics>")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/raylib-library.txt" CONTENT "$<TARGET_FILE:Malvado::raylib>")
endif()
